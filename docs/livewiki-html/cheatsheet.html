Cheatsheet

Quick reference for common UserStore operations

Store Initialization

userStore, err := userstore.NewStore(userstore.NewStoreOptions{
    DB:                 databaseInstance,
    UserTableName:      "user",
    AutomigrateEnabled: true,
    DebugEnabled:       false,
})

User Operations

Create User

user := userstore.NewUser().
    SetStatus(userstore.USER_STATUS_ACTIVE).
    SetFirstName("John").
    SetLastName("Doe").
    SetEmail("test@test.com")

err := user.SetPasswordAndHash("password123")
err = userStore.UserCreate(context.Background(), user)

Find User by ID

user, err := userStore.UserFindByID(context.Background(), userID)

Find User by Email

user, err := userStore.UserFindByEmail(context.Background(), "test@test.com")

List Users

query := userstore.NewUserQuery().
    SetStatus(userstore.USER_STATUS_ACTIVE).
    SetLimit(10)

users, err := userStore.UserList(context.Background(), query)

Update User

user.SetFirstName("Jane")
user.SetEmail("jane@example.com")

err := userStore.UserUpdate(context.Background(), user)

Soft Delete User

err := userStore.UserSoftDelete(context.Background(), user)

Hard Delete User

err := userStore.UserDelete(context.Background(), user)

Count Users

query := userstore.NewUserQuery().
    SetStatus(userstore.USER_STATUS_ACTIVE)

count, err := userStore.UserCount(context.Background(), query)

user.SetRole(userstore.USER_ROLE_ADMINISTRATOR)

Password Operations

Set and Hash Password

Compare Password

Metadata Operations

Set Single Metadata

Get Single Metadata

value := user.GetMeta("customField")

Set Multiple Metadata

Query Builder

Filter by Status

query := userstore.NewUserQuery().
    SetStatus(userstore.USER_STATUS_ACTIVE)

Filter by Email

query := userstore.NewUserQuery().
    SetEmail("test@test.com")

Filter by Email Pattern

query := userstore.NewUserQuery().
    SetEmailLike("%@example.com")

Filter by Name

query := userstore.NewUserQuery().
    SetFirstName("John").
    SetLastName("Doe")

Filter by Multiple Statuses

query := userstore.NewUserQuery().
    SetStatusIn([]string{
        userstore.USER_STATUS_ACTIVE,
        userstore.USER_STATUS_INACTIVE,
    })

Filter by Date Range

query := userstore.NewUserQuery().
    SetCreatedAtGte("2024-01-01").
    SetCreatedAtLte("2024-12-31")

Pagination

query := userstore.NewUserQuery().
    SetLimit(10).
    SetOffset(0)

Sorting

query := userstore.NewUserQuery().
    SetOrderBy("created_at").
    SetSortDirection("DESC")

Include Soft Deleted

query := userstore.NewUserQuery().
    SetSoftDeletedIncluded(true)

Select Specific Columns

query := userstore.NewUserQuery().
    SetColumns([]string{"id", "email", "first_name", "last_name"})

Count Only

query := userstore.NewUserQuery().
    SetCountOnly(true)

Migration

Create Table

err := userStore.MigrateUp(context.Background())

Drop Table

err := userStore.MigrateDown(context.Background())

Migration with Transaction

tx, err := db.Begin()
if err != nil {
    return err
}

err = userStore.MigrateUp(context.Background(), tx)
if err != nil {
    tx.Rollback()
    return err
}

err = tx.Commit()
if err != nil {
    return err
}

Common Patterns

Complete User Creation Flow

// Create user
user := userstore.NewUser().
    SetStatus(userstore.USER_STATUS_ACTIVE).
    SetFirstName("John").
    SetLastName("Doe").
    SetEmail("test@test.com")

// Set password
err := user.SetPasswordAndHash("password123")
if err != nil {
    return err
}

// Save to database
err = userStore.UserCreate(context.Background(), user)
if err != nil {
    return err
}

Authentication Flow

// Find user by email
user, err := userStore.UserFindByEmail(context.Background(), email)
if err != nil {
    return err
}

// Check if user exists and is active
if user == nil || !user.IsActive() {
    return errors.New("invalid credentials")
}

// Verify password
if !user.PasswordCompare(password) {
    return errors.New("invalid credentials")
}

// Authentication successful

Paginated List

page := 1
perPage := 10

query := userstore.NewUserQuery().
    SetStatus(userstore.USER_STATUS_ACTIVE).
    SetLimit(perPage).
    SetOffset((page - 1) * perPage).
    SetOrderBy("created_at").
    SetSortDirection("DESC")

users, err := userStore.UserList(context.Background(), query)
